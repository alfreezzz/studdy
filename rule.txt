rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{userId} {

        // Pengguna yang login dapat membaca profil
        allow read: if request.auth != null;

        // Pengguna hanya boleh membuat profil miliknya
        allow create: if request.auth != null
                        && request.auth.uid == userId
                        && request.resource.data.uid == userId;

        // Pengguna hanya boleh memperbarui profil miliknya
        allow update: if request.auth != null
                        && request.auth.uid == userId
                        && request.resource.data.uid == resource.data.uid;

        // Penghapusan profil belum diizinkan
        allow delete: if false;
    }

    match /studyIntents/{intentId} {

        // Pengguna login dapat membaca intent aktif
        // dan intent miliknya sendiri.
        allow read: if request.auth != null
            && (
            resource.data.status == "active"
            || resource.data.ownerUid == request.auth.uid
            );

        // Membuat intent hanya untuk akun sendiri.
        allow create: if request.auth != null
            && request.resource.data.ownerUid == request.auth.uid
            && request.resource.data.keys().hasAll([
            "ownerUid",
            "type",
            "courseName",
            "topic",
            "description",
            "goal",
            "preferredMode",
            "preferredSchedule",
            "status",
            "createdAt",
            "updatedAt"
            ])
            && request.resource.data.keys().hasOnly([
            "ownerUid",
            "type",
            "courseName",
            "topic",
            "description",
            "goal",
            "preferredMode",
            "preferredSchedule",
            "mentorDirection",
            "status",
            "createdAt",
            "updatedAt"
            ])
            && request.resource.data.type in [
            "study_buddy",
            "project_partner",
            "peer_mentor"
            ]
            && request.resource.data.preferredMode in [
            "online",
            "offline",
            "both"
            ]
            && request.resource.data.preferredSchedule is list
            && request.resource.data.status == "active"
            && request.resource.data.createdAt is timestamp
            && request.resource.data.updatedAt is timestamp
            && (
            !("mentorDirection" in request.resource.data)
            || request.resource.data.mentorDirection in [
                "seeking_mentor",
                "offering_mentorship"
            ]
            );

        // Untuk sementara hanya mengizinkan perubahan status.
        allow update: if request.auth != null
            && request.auth.uid == resource.data.ownerUid
            && request.resource.data.ownerUid == resource.data.ownerUid
            && request.resource.data.diff(resource.data)
            .affectedKeys()
            .hasOnly(["status", "updatedAt"])
            && request.resource.data.status in [
            "active",
            "paused",
            "closed"
            ]
            && request.resource.data.updatedAt is timestamp;

        // Hanya pemilik yang boleh menghapus intent.
        allow delete: if request.auth != null
            && request.auth.uid == resource.data.ownerUid;
    }
  }
}
